🔌 API למפתחים ו-AI Agents

כל הפונקציונליות זמינה דרך REST API נקי ומתועד

// Example: Order via API
const response = await fetch('https://japanika.ai/api/order', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
        message: "I want 2 salmon nigiri and miso soup",
        branch: "tel-aviv-dizengoff"
    })
});

const order = await response.json();
console.log(order); // { orderId: "...", items: [...], total: 89 }
